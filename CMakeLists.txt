cmake_minimum_required(VERSION 3.16)
project(comanchecd_renderer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
option(COMANCHECD_BUILD_DIAGNOSTICS "Build reverse-engineering probes and diagnostic tools" ON)
option(COMANCHECD_BUILD_SDL2_HOST "Build the SDL2 training viewer and audio audition" ON)
option(COMANCHECD_BUILD_OPENXR "Build optional OpenXR runtime admission support" OFF)
if(COMANCHECD_BUILD_OPENXR AND NOT COMANCHECD_BUILD_SDL2_HOST)
    message(FATAL_ERROR "OpenXR runtime support requires the SDL2 host")
endif()

include(cmake/Runtime.cmake)
if(BUILD_TESTING)
    include(cmake/Tests.cmake)
endif()
if(COMANCHECD_BUILD_DIAGNOSTICS)
    include(cmake/Diagnostics.cmake)
endif()
if(COMANCHECD_BUILD_SDL2_HOST)
    include(cmake/SdlHost.cmake)
endif()
